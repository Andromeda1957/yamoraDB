cmake_minimum_required(VERSION 3.16)
project(yamora_conformance LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "SDK tests belong to the parent gate" FORCE)
add_subdirectory(.. sdk)
add_executable(yamora_conformance main.cpp)
target_link_libraries(yamora_conformance PRIVATE yamora::cpp)
target_compile_options(yamora_conformance PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(yamora_http_replica_conformance http_replica.cpp)
target_link_libraries(yamora_http_replica_conformance PRIVATE yamora::cpp)
target_compile_options(yamora_http_replica_conformance PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

add_executable(yamora_operations_conformance operations.cpp)
target_link_libraries(yamora_operations_conformance PRIVATE yamora::cpp)
target_compile_options(yamora_operations_conformance PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
